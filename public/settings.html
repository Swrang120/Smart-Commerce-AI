Dedicated Database

Connect your dedicated Supabase project to persist stores, products, orders, and viral marketing schedules.

📋 Dedicated Supabase SQL Setup Script:

Run this SQL script in your Supabase project SQL Editor to instantiate all 5 tables: 

-- 1. STORES TABLE
CREATE TABLE stores (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    user_id UUID,
    store_name VARCHAR(255) NOT NULL,
    platform VARCHAR(50) NOT NULL,
    api_key TEXT,
    api_secret TEXT,
    access_token TEXT,
    status VARCHAR(50) DEFAULT 'active',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. PRODUCTS TABLE
CREATE TABLE products (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    store_id UUID,
    title VARCHAR(255) NOT NULL,
    category VARCHAR(100),
    cost_price NUMERIC(10, 2) NOT NULL,
    selling_price NUMERIC(10, 2) NOT NULL,
    estimated_profit NUMERIC(10, 2) NOT NULL,
    demand_score INT DEFAULT 0,
    stock_quantity INT DEFAULT 0,
    listing_status VARCHAR(50) DEFAULT 'draft',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. ORDERS & EARNINGS TABLE
CREATE TABLE orders (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    store_id UUID,
    product_id UUID,
    order_number VARCHAR(100) NOT NULL,
    revenue NUMERIC(10, 2) NOT NULL,
    net_profit NUMERIC(10, 2) NOT NULL,
    order_status VARCHAR(50) DEFAULT 'pending',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. SOCIAL MEDIA CONTENT SCHEDULE TABLE
CREATE TABLE social_posts (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    product_id UUID,
    platform VARCHAR(50) NOT NULL,
    video_url TEXT,
    caption TEXT,
    hashtags TEXT,
    post_status VARCHAR(50) DEFAULT 'scheduled',
    published_at TIMESTAMP WITH TIME ZONE
);

-- 5. AI LOGS & DECISIONS
CREATE TABLE ai_logs (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    action_type VARCHAR(100) NOT NULL,
    description TEXT NOT NULL,
    impact_type VARCHAR(50) DEFAULT 'info',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
